namespace = forerunner_ecumene_rates

# Covers starting rulers, existing leaders and pools when a game is initialized.
country_event = {
	id = forerunner_ecumene_rates.1
	hide_window = yes
	is_triggered_only = yes
	trigger = { is_forerunner_ecumene = yes }
	immediate = {
		# The dependency's preset was authored with auth_oligarchic. Convert only that preset/state;
		# player-created Ecumenes may select the custom authority directly at empire creation.
		if = {
			limit = { has_authority = auth_oligarchic }
			set_authority = auth_forerunner_ecumene
		}
		every_owned_leader = { forerunner_assign_rate = yes }
		every_pool_leader = { forerunner_assign_rate = yes }
	}
}

# Works beside the dependency's baseline-trait event; only organic Forerunners are considered.
country_event = {
	id = forerunner_ecumene_rates.2
	hide_window = yes
	is_triggered_only = yes
	trigger = { from = { is_organic_forerunner_leader = yes } }
	immediate = { from = { forerunner_assign_rate = yes } }
}
