# Stable compatibility hooks for optional submods. Leader-scoped triggers begin with is_/has_.
is_forerunner_ecumene = { has_civic = civic_forerunner_ecumene }
is_forerunner_oligarchic_government = {
	OR = {
		has_authority = auth_forerunner_ecumene
		has_authority = auth_oligarchic
	}
	has_civic = civic_forerunner_ecumene
}
is_organic_forerunner_leader = {
	species = { has_trait = trait_forerunner_species }
}
has_forerunner_rate = {
	OR = {
		has_trait = leader_trait_forerunner_rate_builder
		has_trait = leader_trait_forerunner_rate_builder_security
		has_trait = leader_trait_forerunner_rate_miner
		has_trait = leader_trait_forerunner_rate_lifeworker
		has_trait = leader_trait_forerunner_rate_juridical
		has_trait = leader_trait_forerunner_rate_engineer
		has_trait = leader_trait_forerunner_rate_warrior_servant
		has_trait = leader_trait_forerunner_rate_theoretical
		has_trait = leader_trait_forerunner_rate_historian
		has_trait = leader_trait_forerunner_rate_weaver
		has_trait = leader_trait_forerunner_rate_speaker
		has_trait = leader_trait_forerunner_rate_interpreter
		has_trait = leader_trait_forerunner_rate_promethean
	}
}
